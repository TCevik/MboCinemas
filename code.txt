// Wacht tot de volledige HTML-pagina is geladen voordat JavaScript wordt uitgevoerd
document.addEventListener('DOMContentLoaded', function () {
    // Sleutel waaronder een niet-afgeronde reservering in localStorage wordt bewaard
    const DRAFT_KEY = 'mbo_cinemas_reservation_draft';

    // 1. CONCEPT WISSEN NA BEVESTIGING
    // Als de reservering is bevestigd (reserve.php), verwijder dan het opgeslagen concept
    const confirmedBox = document.getElementById('reservation-confirmed');
    if (confirmedBox && confirmedBox.dataset.clearDraft === 'true') {
        localStorage.removeItem(DRAFT_KEY);
    }

    // 2. BANNER VOOR NIET-AFGERONDE RESERVERING (index.php)
    // Controleert of er nog een concept in localStorage staat en toont zo ja de hervat-banner
    const banner = document.getElementById('unfinished-reservation-banner');
    const resumeLink = document.getElementById('resume-reservation-link');
    const discardBtn = document.getElementById('discard-draft-btn');
    if (banner && resumeLink) {
        const saved = localStorage.getItem(DRAFT_KEY);
        if (saved) {
            try {
                const draft = JSON.parse(saved);
                if (draft && draft.screening_id) {
                    resumeLink.href = '/reserve.php?screening_id=' + encodeURIComponent(draft.screening_id);
                    banner.classList.remove('hidden');
                }
            } catch (e) {
                // Verwijder corrupte data uit localStorage
                localStorage.removeItem(DRAFT_KEY);
            }
        }
        // Knop om het opgeslagen concept handmatig te wissen
        if (discardBtn) {
            discardBtn.addEventListener('click', function () {
                localStorage.removeItem(DRAFT_KEY);
                banner.classList.add('hidden');
            });
        }
    }

    // 3. LIVE FILTEREN VAN FILMS (index.php)
    // Filtert de filmkaarten direct in de browser zonder de pagina opnieuw te laden
    const searchInput = document.getElementById('filter-search');
    const genreSelect = document.getElementById('filter-genre');
    const locationSelect = document.getElementById('filter-location');
    const dateInput = document.getElementById('filter-date');
    const movieCards = document.querySelectorAll('.movie-card');

    function filterMovies() {
        const search = searchInput ? searchInput.value.toLowerCase().trim() : '';
        const genre = genreSelect ? genreSelect.value : '';
        const location = locationSelect ? locationSelect.value : '';
        const date = dateInput ? dateInput.value : '';

        movieCards.forEach(function (card) {
            const matchTitle = search === '' || card.dataset.title.includes(search);
            const matchGenre = genre === '' || card.dataset.genre === genre;
            const matchLocation = location === '' || card.dataset.location === location;
            const matchDate = date === '' || card.dataset.date === date;

            // Toon de kaart alleen als alle actieve filters overeenkomen
            if (matchTitle && matchGenre && matchLocation && matchDate) {
                card.classList.remove('hidden');
            } else {
                card.classList.add('hidden');
            }
        });
    }

    if (movieCards.length > 0) {
        if (searchInput) searchInput.addEventListener('input', filterMovies);
        if (genreSelect) genreSelect.addEventListener('change', filterMovies);
        if (locationSelect) locationSelect.addEventListener('change', filterMovies);
        if (dateInput) dateInput.addEventListener('change', filterMovies);
    }

    // 4. RESERVERINGSFORMULIER: CONCEPT OPSLAAN, TOTAALPRIJS & VALIDATIE (reserve.php)
    const reservationForm = document.getElementById('reservation-form');
    if (reservationForm) {
        const screeningId = reservationForm.dataset.screeningId;
        const nameInput = document.getElementById('customer_name');
        const emailInput = document.getElementById('customer_email');
        const phoneInput = document.getElementById('customer_phone');
        const seatsInput = document.getElementById('seats');
        const priceDisplay = document.getElementById('total-price-display');

        // Herstel eerder ingevulde gegevens als het concept bij deze voorstelling hoort
        const saved = localStorage.getItem(DRAFT_KEY);
        if (saved) {
            try {
                const draft = JSON.parse(saved);
                if (draft && String(draft.screening_id) === String(screeningId)) {
                    if (draft.name && nameInput.value === '') nameInput.value = draft.name;
                    if (draft.email && emailInput.value === '') emailInput.value = draft.email;
                    if (draft.phone) phoneInput.value = draft.phone;
                    if (draft.seats) seatsInput.value = draft.seats;
                }
            } catch (e) {
                localStorage.removeItem(DRAFT_KEY);
            }
        }

        // Berekent en toont de totaalprijs live (aantal kaartjes * â‚¬11,50)
        function updateTotalPrice() {
            const count = parseInt(seatsInput.value, 10) || 0;
            const total = (count * 11.50).toFixed(2).replace('.', ',');
            priceDisplay.textContent = '\u20AC' + total;
        }

        // Slaat de huidige invoer tussentijds op in localStorage
        function saveDraft() {
            const draft = {
                screening_id: screeningId,
                name: nameInput.value,
                email: emailInput.value,
                phone: phoneInput.value,
                seats: seatsInput.value
            };
            localStorage.setItem(DRAFT_KEY, JSON.stringify(draft));
        }

        // Luister naar wijzigingen in alle invoervelden
        [nameInput, emailInput, phoneInput, seatsInput].forEach(function (input) {
            input.addEventListener('input', function () {
                saveDraft();
                if (input === seatsInput) {
                    updateTotalPrice();
                }
            });
        });

        updateTotalPrice();

        // Client-side validatie en bevestigingspop-up bij het versturen van het formulier
        reservationForm.addEventListener('submit', function (e) {
            const seats = parseInt(seatsInput.value, 10);
            const maxSeats = parseInt(seatsInput.max, 10);
            if (nameInput.value.trim() === '' || emailInput.value.trim() === '' || phoneInput.value.trim() === '') {
                e.preventDefault();
                alert('Vul alle verplichte velden in.');
                return;
            }
            if (isNaN(seats) || seats < 1 || seats > maxSeats) {
                e.preventDefault();
                alert('Kies een geldig aantal kaartjes (maximaal ' + maxSeats + ').');
                return;
            }
            if (!confirm('Weet u zeker dat u ' + seats + ' kaartje(s) wilt reserveren?')) {
                e.preventDefault();
            }
        });
    }

    // 5. BEVESTIGINGSPOP-UP VOOR VERWIJDEREN / ANNULEREN
    // Voorkomt dat bezoekers of medewerkers per ongeluk iets verwijderen of annuleren
    document.querySelectorAll('.confirm-cancel').forEach(function (form) {
        form.addEventListener('submit', function (e) {
            if (!confirm('Weet u zeker dat u deze actie wilt uitvoeren?')) {
                e.preventDefault();
            }
        });
    });
});

class Movie {

        return $stmt->fetchAll();
    }

    public function getMovieById($id) {
        $stmt = $this->db->prepare("SELECT * FROM movies WHERE id = ?");
        $stmt->execute([$id]);
        return $stmt->fetch();
    }

    public function getGenres() {
        $stmt = $this->db->query("SELECT DISTINCT genre FROM movies ORDER BY genre ASC");
        return $stmt->fetchAll(PDO::FETCH_COLUMN);
    }

    public function getLocations() {
        $stmt = $this->db->query("SELECT * FROM locations ORDER BY name ASC");
        return $stmt->fetchAll();
    }

    public function getHalls() {
        $sql = "SELECT h.*, l.name AS location_name
                FROM halls h
                JOIN locations l ON h.location_id = l.id
                ORDER BY l.name ASC, h.name ASC";
        $stmt = $this->db->query($sql);
        return $stmt->fetchAll();
    }

    public function getFilteredScreenings($search = '', $genre = '', $locationId = '', $date = '') {
        $sql = "SELECT s.id AS screening_id, s.screening_time,
                       m.id AS movie_id, m.title, m.description, m.genre, m.duration_minutes,
                       h.id AS hall_id, h.name AS hall_name, h.capacity,
                       l.id AS location_id, l.name AS location_name, l.city,
                       COALESCE(SUM(CASE WHEN r.status = 'actief' THEN r.seats ELSE 0 END), 0) AS booked_seats
                FROM screenings s
                JOIN movies m ON s.movie_id = m.id
                JOIN halls h ON s.hall_id = h.id
                JOIN locations l ON h.location_id = l.id
                LEFT JOIN reservations r ON r.screening_id = s.id
                WHERE 1=1";
        $params = [];

        if ($search !== '') {
            $sql .= " AND (m.title LIKE ? OR m.description LIKE ?)";
            $params[] = '%' . $search . '%';
            $params[] = '%' . $search . '%';
        }
        if ($genre !== '') {
            $sql .= " AND m.genre = ?";
            $params[] = $genre;
        }
        if ($locationId !== '') {
            $sql .= " AND l.id = ?";
            $params[] = $locationId;
        }
        if ($date !== '') {
            $sql .= " AND DATE(s.screening_time) = ?";
            $params[] = $date;
        }

        $sql .= " GROUP BY s.id ORDER BY s.screening_time ASC";

        $stmt = $this->db->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }

    public function getScreeningById($id) {
        $sql = "SELECT s.id AS screening_id, s.screening_time,
                       m.id AS movie_id, m.title, m.description, m.genre, m.duration_minutes,
                       h.id AS hall_id, h.name AS hall_name, h.capacity,
                       l.id AS location_id, l.name AS location_name, l.city,
                       COALESCE(SUM(CASE WHEN r.status = 'actief' THEN r.seats ELSE 0 END), 0) AS booked_seats
                FROM screenings s
                JOIN movies m ON s.movie_id = m.id
                JOIN halls h ON s.hall_id = h.id

<?php
class Reservation {
    private $db;

    public function __construct($pdo) {
        $this->db = $pdo;
    }

    public function create($screeningId, $name, $email, $phone, $seats) {
        $code = strtoupper(substr(md5(uniqid((string)mt_rand(), true)), 0, 8));
        $stmt = $this->db->prepare("INSERT INTO reservations (reservation_code, screening_id, customer_name, customer_email, customer_phone, seats, status) VALUES (?, ?, ?, ?, ?, ?, 'actief')");
        $stmt->execute([$code, (int)$screeningId, $name, $email, $phone, (int)$seats]);
        return $code;
    }

    public function getByEmail($email) {
        $sql = "SELECT r.*, s.screening_time, m.title AS movie_title, h.name AS hall_name, h.capacity, l.name AS location_name
                FROM reservations r
                JOIN screenings s ON r.screening_id = s.id
                JOIN movies m ON s.movie_id = m.id
                JOIN halls h ON s.hall_id = h.id
                JOIN locations l ON h.location_id = l.id
                WHERE r.customer_email = ?
                ORDER BY s.screening_time DESC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$email]);
        return $stmt->fetchAll();
    }

    public function getById($id) {
        $stmt = $this->db->prepare("SELECT * FROM reservations WHERE id = ?");
        $stmt->execute([(int)$id]);
        return $stmt->fetch();
    }

    public function updateSeats($id, $seats) {
        $stmt = $this->db->prepare("UPDATE reservations SET seats = ? WHERE id = ? AND status = 'actief'");
        return $stmt->execute([(int)$seats, (int)$id]);
    }

    public function cancel($id) {
        $stmt = $this->db->prepare("UPDATE reservations SET status = 'geannuleerd' WHERE id = ?");
        return $stmt->execute([(int)$id]);
    }

    public function getAllByLocation($locationId = '') {
        $sql = "SELECT r.*, s.screening_time, m.title AS movie_title, h.name AS hall_name, h.capacity, l.id AS location_id, l.name AS location_name
                FROM reservations r
                JOIN screenings s ON r.screening_id = s.id
                JOIN movies m ON s.movie_id = m.id
                JOIN halls h ON s.hall_id = h.id
                JOIN locations l ON h.location_id = l.id
                WHERE 1=1";
        $params = [];

        if ($locationId !== '') {
            $sql .= " AND l.id = ?";
            $params[] = (int)$locationId;
        }

        $sql .= " ORDER BY l.name ASC, s.screening_time ASC, r.created_at DESC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }
}

class User {
    private $db;

    public function __construct($pdo) {
        $this->db = $pdo;
    }

    public function login($email, $password) {
        $stmt = $this->db->prepare("SELECT * FROM users WHERE email = ?");
        $stmt->execute([$email]);
        $user = $stmt->fetch();

        if ($user && password_verify($password, $user['password'])) {
            return $user;
        }
        return false;
    }

    public function getById($id) {
        $stmt = $this->db->prepare("SELECT id, name, email, role FROM users WHERE id = ?");
        $stmt->execute([(int)$id]);
        return $stmt->fetch();
    }

    public function updateAccount($id, $name, $email, $newPassword = '') {
        if ($newPassword !== '') {
            $hash = password_hash($newPassword, PASSWORD_DEFAULT);
            $stmt = $this->db->prepare("UPDATE users SET name = ?, email = ?, password = ? WHERE id = ?");
            return $stmt->execute([$name, $email, $hash, (int)$id]);
        }
        $stmt = $this->db->prepare("UPDATE users SET name = ?, email = ? WHERE id = ?");
        return $stmt->execute([$name, $email, (int)$id]);
    }
}

        }

        if ($action === 'add_movie') {
            if ($title !== '' && $description !== '' && $genre !== '' && $duration > 0) {
                $movieModel->addMovie($title, $description, $genre, $duration);
                $message = 'Film toegevoegd.';
            } else {
                $error = 'Vul alle verplichte filmvelden in.';
            }
        } else {
            $movieId = 0;
            if (isset($_POST['movie_id'])) {
                $movieId = $_POST['movie_id'];
            }
            if ($movieId > 0 && $title !== '' && $description !== '' && $genre !== '' && $duration > 0) {
                $movieModel->updateMovie($movieId, $title, $description, $genre, $duration);
                $message = 'Film bijgewerkt.';
            } else {
                $error = 'Ongeldige filmgegevens.';
            }
        }
    } elseif ($action === 'delete_movie') {
        $movieId = 0;
        if (isset($_POST['movie_id'])) {
            $movieId = $_POST['movie_id'];
        }
        if ($movieId > 0) {
            $movieModel->deleteMovie($movieId);
            $message = 'Film verwijderd.';
        }
    } elseif ($action === 'add_screening') {
        $movieId = 0;
        if (isset($_POST['movie_id'])) {
            $movieId = $_POST['movie_id'];
        }
        $hallId = 0;
        if (isset($_POST['hall_id'])) {
            $hallId = $_POST['hall_id'];
        }
        $screeningTime = '';
        if (isset($_POST['screening_time'])) {
            $screeningTime = trim($_POST['screening_time']);
        }

        if ($movieId > 0 && $hallId > 0 && $screeningTime !== '') {
            $movieModel->addScreening($movieId, $hallId, str_replace('T', ' ', $screeningTime));
            $message = 'Voorstelling ingepland.';
        } else {
            $error = 'Vul alle velden voor de voorstelling in.';
        }
    } elseif ($action === 'delete_screening') {
        $screeningId = 0;
        if (isset($_POST['screening_id'])) {
            $screeningId = $_POST['screening_id'];
        }
        if ($screeningId > 0) {
            $movieModel->deleteScreening($screeningId);
            $message = 'Voorstelling verwijderd.';
        }
    } elseif ($action === 'update_reservation') {
        $reservationId = 0;
        if (isset($_POST['reservation_id'])) {
            $reservationId = $_POST['reservation_id'];
        }
        $seats = 0;
        if (isset($_POST['seats'])) {
            $seats = (int)$_POST['seats'];
        }
        if ($reservationId > 0 && $seats >= 1 && $seats <= 20) {
            $reservationModel->updateSeats($reservationId, $seats);
            $message = 'Reservering bijgewerkt.';
        } else {
            $error = 'Ongeldig aantal kaartjes.';
        }
    } elseif ($action === 'cancel_reservation') {
        $reservationId = 0;
        if (isset($_POST['reservation_id'])) {
            $reservationId = $_POST['reservation_id'];
        }
        if ($reservationId > 0) {
            $reservationModel->cancel($reservationId);
            $message = 'Reservering geannuleerd.';
        }
    } elseif ($action === 'update_account') {
        $name = '';
        if (isset($_POST['name'])) {
            $name = trim($_POST['name']);
        }
        $email = '';
        if (isset($_POST['email'])) {
            $email = trim($_POST['email']);
        }
        $newPassword = '';
        if (isset($_POST['new_password'])) {
            $newPassword = $_POST['new_password'];
        }

            $userModel->updateAccount($_SESSION['user_id'], $name, $email, $newPassword);
            $_SESSION['user_name'] = $name;
            $message = 'Accountgegevens bijgewerkt.';
        } else {
            $error = 'Vul een geldige naam en e-mailadres in.';
        }
    }

CREATE DATABASE IF NOT EXISTS phples CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE phples;

-- Tabel voor de verschillende bioscooplocaties
CREATE TABLE IF NOT EXISTS locations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    city VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS halls (
    id INT AUTO_INCREMENT PRIMARY KEY,
    location_id INT NOT NULL,
    name VARCHAR(50) NOT NULL,
    capacity INT NOT NULL DEFAULT 60,
    FOREIGN KEY (location_id) REFERENCES locations(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS movies (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(150) NOT NULL,
    description TEXT NOT NULL,
    genre VARCHAR(50) NOT NULL,
    duration_minutes INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS screenings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    movie_id INT NOT NULL,
    hall_id INT NOT NULL,
    screening_time DATETIME NOT NULL,
    FOREIGN KEY (movie_id) REFERENCES movies(id) ON DELETE CASCADE,
    FOREIGN KEY (hall_id) REFERENCES halls(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS reservations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    reservation_code VARCHAR(12) NOT NULL UNIQUE,
    screening_id INT NOT NULL,
    customer_name VARCHAR(100) NOT NULL,
    customer_email VARCHAR(150) NOT NULL,
    customer_phone VARCHAR(20) NOT NULL,
    seats INT NOT NULL DEFAULT 1,
    status ENUM('actief', 'geannuleerd') DEFAULT 'actief',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (screening_id) REFERENCES screenings(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role VARCHAR(50) DEFAULT 'medewerker',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

<?php
require_once __DIR__ . '/classes/Database.php';

// Als de medewerker al is ingelogd, stuur direct door naar het beheerpaneel
if (isLoggedIn()) {
    redirect('/admin.php');
}

$database = new Database();
$db = $database->getConnection();
$userModel = new User($db);
$error = '';

// Verwerk het inlogformulier bij een POST-verzoek
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $email = '';
    if (isset($_POST['email'])) {
        $email = trim($_POST['email']);
    }

    $password = '';
    if (isset($_POST['password'])) {
        $password = $_POST['password'];
    }

    // Controleer of beide velden zijn ingevuld
    if ($email === '' || $password === '') {
        $error = 'Vul zowel e-mailadres als wachtwoord in.';
    } else {
        // Controleer de inloggegevens via de User-class
        $user = $userModel->login($email, $password);
        if ($user) {
            // Sla gebruikersgegevens op in de sessie en stuur door naar admin.php
            $_SESSION['user_id'] = $user['id'];
            $_SESSION['user_name'] = $user['name'];
            redirect('/admin.php');
        } else {
            $error = 'Ongeldige inloggegevens.';
        }
    }
}

require_once __DIR__ . '/includes/header.php';
?>

    <h1>Medewerker Inloggen</h1>
</section>

<?php
require_once __DIR__ . '/classes/Database.php';

$database = new Database();
$db = $database->getConnection();
$movieModel = new Movie($db);
$reservationModel = new Reservation($db);

// Haal het voorstellings-ID op uit de URL (GET) of uit het verzonden formulier (POST)
$screeningId = 0;
if (isset($_GET['screening_id'])) {
    $screeningId = $_GET['screening_id'];
} elseif (isset($_POST['screening_id'])) {
    $screeningId = $_POST['screening_id'];
}

// Haal de voorstelling op; als deze niet bestaat, stuur terug naar de homepagina
$screening = $movieModel->getScreeningById($screeningId);
if (!$screening) {
    redirect('/index.php');
}

// Bereken vrije stoelen en formatteer datum/tijd en ticketprijs
$availableSeats = $screening['capacity'] - $screening['booked_seats'];
$formattedDateTime = date('d-m-Y H:i', strtotime($screening['screening_time']));
$formattedTicketPrice = number_format(TICKET_PRICE, 2, ',', '.');
$errors = [];
$confirmationCode = '';

// Vul naam en e-mail alvast in als deze eerder in een cookie zijn opgeslagen
$customerName = '';
if (isset($_COOKIE['customer_name'])) {
    $customerName = $_COOKIE['customer_name'];
}

$customerEmail = '';
if (isset($_COOKIE['customer_email'])) {
    $customerEmail = $_COOKIE['customer_email'];
}

$customerPhone = '';
$seats = 1;

// Verwerk het reserveringsformulier bij een POST-verzoek
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['customer_name'])) {
        $customerName = trim($_POST['customer_name']);
    }
    if (isset($_POST['customer_email'])) {
        $customerEmail = trim($_POST['customer_email']);
    }
    if (isset($_POST['customer_phone'])) {
        $customerPhone = trim($_POST['customer_phone']);
    }
    if (isset($_POST['seats'])) {
        $seats = (int)$_POST['seats'];
    }

    // Server-side validatie van alle invoervelden
    if ($customerName === '') {
        $errors[] = 'Vul uw naam in.';
    }
    if (!filter_var($customerEmail, FILTER_VALIDATE_EMAIL)) {
        $errors[] = 'Vul een geldig e-mailadres in.';
    }
    if ($customerPhone === '') {
        $errors[] = 'Vul uw telefoonnummer in.';
    }
    if ($seats < 1 || $seats > $availableSeats) {
        $errors[] = 'Kies een geldig aantal kaartjes (maximaal ' . $availableSeats . ').';
    }

    // Als er geen fouten zijn: sla reservering op en bewaar klantgegevens 30 dagen in cookies
    if (empty($errors)) {
        $confirmationCode = $reservationModel->create($screeningId, $customerName, $customerEmail, $customerPhone, $seats);
        setcookie('customer_name', $customerName, time() + (30 * 24 * 60 * 60), '/');
        setcookie('customer_email', $customerEmail, time() + (30 * 24 * 60 * 60), '/');
    }
}

$formattedTotalPrice = number_format($seats * TICKET_PRICE, 2, ',', '.');

require_once __DIR__ . '/includes/header.php';
?>

<section class="page-header">
    <h1>Kaartjes Reserveren</h1>
    <a href="/index.php" class="btn btn-secondary">Terug naar overzicht</a>
</section>

<!-- Toon bevestiging als de reservering succesvol is opgeslagen; data-clear-draft="true" wist het concept in localStorage -->
<?php if ($confirmationCode !== ''): ?>
    <div class="alert alert-success" id="reservation-confirmed" data-clear-draft="true">
        <h2>Reservering Bevestigd!</h2>
        <p>Bedankt <strong><?php echo sanitize($customerName); ?></strong>. Uw reservering is opgeslagen.</p>
        <p><strong>Reserveringscode:</strong> <?php echo sanitize($confirmationCode); ?></p>
        <p><strong>Film:</strong> <?php echo sanitize($screening['title']); ?></p>
        <p><strong>Locatie:</strong> <?php echo sanitize($screening['location_name']); ?> (<?php echo sanitize($screening['hall_name']); ?>)</p>
        <p><strong>Tijdstip:</strong> <?php echo $formattedDateTime; ?></p>
        <p><strong>Aantal kaartjes:</strong> <?php echo $seats; ?> (&euro;<?php echo $formattedTotalPrice; ?>)</p>
        <a href="/my-reservations.php?email=<?php echo urlencode($customerEmail); ?>" class="btn">Bekijk mijn reserveringen</a>
    </div>
<?php else: ?>
    <!-- Samenvatting van de gekozen voorstelling -->
    <div class="card screening-summary">
        <h2><?php echo sanitize($screening['title']); ?></h2>
        <p><strong>Locatie:</strong> <?php echo sanitize($screening['location_name']); ?> - <?php echo sanitize($screening['hall_name']); ?></p>
        <p><strong>Datum &amp; Tijd:</strong> <?php echo $formattedDateTime; ?></p>
        <p><strong>Beschikbare stoelen:</strong> <?php echo $availableSeats; ?></p>
        <p><strong>Prijs per kaartje:</strong> &euro;<?php echo $formattedTicketPrice; ?></p>
    </div>

    <!-- Toon eventuele validatiefouten -->
    <?php if (!empty($errors)): ?>
        <div class="alert alert-danger">
            <ul>
                <?php foreach ($errors as $error): ?>
                    <li><?php echo sanitize($error); ?></li>
                <?php endforeach; ?>
            </ul>
        </div>
    <?php endif; ?>

    <!-- Reserveringsformulier; main.js slaat wijzigingen tussentijds op in localStorage en berekent de totaalprijs live -->
    <form method="POST" action="/reserve.php" id="reservation-form" class="card form-card" data-screening-id="<?php echo $screeningId; ?>">
        <input type="hidden" name="screening_id" value="<?php echo $screeningId; ?>">

        <div class="form-group">

        </div>
